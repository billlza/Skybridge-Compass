   [ Fr( ~sk ) ]
  --[ RegisterIdentity( $A, pk(~sk) ), OnlyOnceIdentity( $A ) ]->
   [ !Ltk( $A, ~sk ), !Pk( $A, pk(~sk) ), Out( pk(~sk) ) ]

  /* has exactly the trivial AC variant */

rule (modulo E) Register_KEMKey:
   [ Fr( ~kemsk ) ]
  --[ RegisterKEM( $A, kempk(~kemsk) ), OnlyOnceKEM( $A ) ]->
   [
   !KemLtk( $A, ~kemsk ), !KemPk( $A, kempk(~kemsk) ), Out( kempk(~kemsk) )
   ]

  /* has exactly the trivial AC variant */

restriction identity_once:
  "∀ A #i #j.
    ((OnlyOnceIdentity( A ) @ #i) ∧ (OnlyOnceIdentity( A ) @ #j)) ⇒
    (#i = #j)"
  // safety formula

restriction kem_once:
  "∀ A #i #j.
    ((OnlyOnceKEM( A ) @ #i) ∧ (OnlyOnceKEM( A ) @ #j)) ⇒ (#i = #j)"
  // safety formula
